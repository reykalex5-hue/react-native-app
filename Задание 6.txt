imporф { useState, useEffect } from 'react';
import { View, Text, TextInput, StyleSheet } from 'react-native';
import * as Location from 'expo-location';

type Weather = {
  city: string;
  temp: number;
  description: string;
};

function describe(code: number): string {
  if (code === 0) return 'Klarvær';
  if (code <= 3) return 'Skyet';
  if (code === 45 || code === 48) return 'Tåke';
  if (code >= 51 && code <= 67) return 'Regn';
  if (code >= 71 && code <= 77) return 'Snø';
  if (code >= 80 && code <= 82) return 'Regnbyger';
  if (code >= 95) return 'Tordenvær';
  return 'Ukjent';
}

function backgroundFor(description: string): string {
  switch (description) {
    case 'Klarvær': return '#2f80ed';
    case 'Skyet': return '#7f8c8d';
    case 'Tåke': return '#b2bec3';
    case 'Regn':
    case 'Regnbyger': return '#34495e';
    case 'Snø': return '#dfe6e9';
    case 'Tordenvær': return '#2d1b4e';
    default: return '#000000';
  }
}

export default function Index() {
  const [query, setQuery] = useState('');
  const [weather, setWeather] = useState<Weather | null>(null);
  const [error, setError] = useState('');

  async function loadWeather(lat: number, lon: number, city: string) {
    const res = await fetch(
      `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=temperature_2m,weather_code`
    );
    const w = await res.json();
    setWeather({
      city,
      temp: w.current.temperature_2m,
      description: describe(w.current.weather_code),
    });
  }

  async function search() {
    try {
      setError('');
      const geoRes = await fetch(
        `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(query)}&count=1`
      );
      const geo = await geoRes.json();
      if (!geo.results || geo.results.length === 0) {
        setError('Fant ikke stedet');
        return;
      }
      const place = geo.results[0];
      await loadWeather(place.latitude, place.longitude, place.name);
    } catch (e) {
      setError('Noe gikk galt');
    }
  }

  useEffect(() => {
    (async () => {
      try {
        const { status } = await Location.requestForegroundPermissionsAsync();
        if (status !== 'granted') return;
        const pos = await Location.getCurrentPositionAsync({});
        const places = await Location.reverseGeocodeAsync(pos.coords);
        const name = places[0]?.city ?? places[0]?.region ?? 'Min posisjon';
        await loadWeather(pos.coords.latitude, pos.coords.longitude, name);
      } catch (e) {
        setError('Kunne ikke finne posisjonen');
      }
    })();
  }, []);

  const background = weather ? backgroundFor(weather.description) : '#1e272e';

  return (
    <View style={[styles.container, { backgroundColor: background }]}>
      {weather && (
        <View style={styles.result}>
          <Text style={styles.city}>{weather.city}</Text>
          <Text style={styles.description}>{weather.description}</Text>
          <Text style={[styles.temp, weather.temp < 0 && styles.cold]}>
            {Math.round(weather.temp)}°
          </Text>
        </View>
      )}

      {error !== '' && <Text style={styles.error}>{error}</Text>}

      <TextInput
        style={styles.input}
        placeholder="Search any city"
        placeholderTextColor="#ddd"
        value={query}
        onChangeText={setQuery}
        onSubmitEditing={search}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', padding: 30 },
  result: { alignItems: 'center', marginBottom: 30 },
  city: { color: 'white', fontSize: 40 },
  description: { color: 'white', fontSize: 18, marginTop: 5 },
  temp: { color: 'white', fontSize: 50, marginTop: 10 },
  cold: { color: 'red' },
  error: { color: '#ffb3b3', textAlign: 'center', marginBottom: 15 },
  input: { backgroundColor: 'rgba(255,255,255,0.25)', color: 'white', padding: 12, borderRadius: 5 },
});
