import React, { useState, useEffect } from 'react';
import {
  StyleSheet,
  Text,
  View,
  TouchableOpacity,
  FlatList,
  TextInput,
  SafeAreaView,
  StatusBar
} from 'react-native';

type LogItem = {
  id: string;
  title: string;
  time: string;
};

export default function App() {
  const [task, setTask] = useState('');
  const [sec, setSec] = useState(0);
  const [active, setActive] = useState(false);
  const [logs, setLogs] = useState<LogItem[]>([]);

  useEffect(() => {
    if (!active) return;

    const timer = setInterval(() => setSec((s) => s + 1), 1000);
    return () => clearInterval(timer);
  }, [active]);

  const fmt = (s: number) => {
    const hrs = Math.floor(s / 3600);
    const mins = Math.floor((s % 3600) / 60);
    const secs = s % 60;
    return [hrs, mins, secs]
      .map((v) => (v < 10 ? '0' + v : v))
      .join(':');
  };

  const save = () => {
    if (!sec) return;
    setLogs((prev) => [
      { id: Date.now().toString(), title: task || 'Uten navn', time: fmt(sec) },
      ...prev,
    ]);
    setActive(false);
    setSec(0);
    setTask('');
  };

  return (
    <SafeAreaView style={s.container}>
      <StatusBar barStyle="light-content" />
      
      <Text style={s.headerTitle}>Tidsmåler</Text>

      {/* Timer display & Card */}
      <View style={s.timerCard}>
        <TextInput
          style={s.input}
          placeholder="Hva jobber du med?..."
          placeholderTextColor="#6e6e73"
          value={task}
          onChangeText={setTask}
        />
        
        <Text style={s.timerText}>{fmt(sec)}</Text>

        <View style={s.row}>
          <TouchableOpacity
            activeOpacity={0.8}
            style={[s.btn, active ? s.pauseBtn : s.startBtn]}
            onPress={() => setActive(!active)}
          >
            <Text style={s.btnTxt}>{active ? 'Paus' : 'Start'}</Text>
          </TouchableOpacity>

          <TouchableOpacity
            activeOpacity={0.8}
            style={[s.btn, s.saveBtn]}
            onPress={save}
          >
            <Text style={s.btnTxt}>Lagre</Text>
          </TouchableOpacity>

          <TouchableOpacity
            activeOpacity={0.8}
            style={[s.btn, s.resetBtn]}
            onPress={() => {
              setActive(false);
              setSec(0);
            }}
          >
            <Text style={s.btnTxt}>Nullstill</Text>
          </TouchableOpacity>
        </View>
      </View>

      {/* Log list */}
      <Text style={s.sectionTitle}>Tidligere økter</Text>

      <FlatList
        data={logs}
        keyExtractor={(item) => item.id}
        contentContainerStyle={{ paddingBottom: 20 }}
        showsVerticalScrollIndicator={false}
        renderItem={({ item }) => (
          <View style={s.historyCard}>
            <View style={s.historyLeft}>
              <View style={s.dot} />
              <Text style={s.historyTitle}>{item.title}</Text>
            </View>
            <Text style={s.historyTime}>{item.time}</Text>
          </View>
        )}
      />
    </SafeAreaView>
  );
}
